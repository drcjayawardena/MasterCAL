/****************************************************** * MasterCAL Pro * Code.gs (v3 — per-user Sheet copies) * * Multi-user model: * - The spreadsheet this script is bound to is the MASTER * (template). } catch (e) {} } const copy = DriveApp.getFileById(MASTER_ID) .makeCopy("MasterCAL — user " + userId, getUserCopiesFolder_()); props.setProperty(key, copy.getId()); return SpreadsheetApp.openById(copy.getId()); } finally { lock.releaseLock(); } } /*==================================================== SHEET / RANGE HELPERS (operate on the USER'S copy) ====================================================*/ function ssActive_() { return CURRENT_SS || SpreadsheetApp.getActiveSpreadsheet(); } function sh() { const sheet = ssActive_().getSheetByName(SHEET_NAME); if (!sheet) throw new Error("Sheet '" + SHEET_NAME + "' not found."); return sheet; } /* Range by CFG key: named range first (on the user's copy), A1 fallback. */ function rng(key) { const named = ssActive_().getRangeByName(key); if (!ch) return []; const data = ch.getRange("A2:B30").getDisplayValues(); const out = []; for (let i = 0; i < data.length; i++) { if (String(data[i][0]).trim().toUpperCase() === "RMV CHARGES" && String(data[i][1]).trim() !== "") { out.push(data[i][1]); } } return out; } /* Set an RMV charge type dropdown (J13 or J14) and recompute. */ function setRmvType(value, cell) { const target = (cell === "J13" || cell === "J14") ? cell : "J14"; sh().getRange(target).setValue(value == null ? } function getOutputs() { const out = getOutputsLite(); /* Dynamic field labels A5:A10 -> facility, bic, insurance, stamp, irr, period */ out.labels = sh().getRange("A5:A10").getDisplayValues().flat(); return out; } function getSchedule() { const v = rng("SCHED_BLOCK").getDisplayValues(); /* F3:H14 */ const rows = []; for (let i = 0; i < 12; i++) rows.push({ from: v[i][0], to: v[i][1], amount: v[i][2] }); return rows; } function toColumn_() { return rng("SCHED_TO").getDisplayValues().flat(); } function getRentalSchedule() { return { success: true, rows: rng("RENTAL_SCHEDULE").getDisplayValues() }; } function clearInputs_() { rng("IN_PRODUCT_TYPE").clearContent(); rng("IN_MAIN").clearContent(); rng("IN_SCHEDULE_TYPE").clearContent(); rng("SCHED_FROM").clearContent(); rng("SCHED_AMOUNT").clearContent(); } /*==================================================== APP LOAD (one round trip) ====================================================*/ function initApp() { clearInputs_(); SpreadsheetApp.flush(); return { success: true, outputs: getOutputs(), schedule: getSchedule() }; } /*==================================================== WRITES (each on the user's OWN copy — no shared lock needed) ====================================================*/ function saveProductType(productType) { rng("IN_PRODUCT_TYPE").setValue(productType == null ?
